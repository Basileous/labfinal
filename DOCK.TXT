from flask import Flask, jsonify, request
from flask_cors import CORS

app = Flask(__name__)
CORS(app)

students = [
    {'id': 1, 'name': 'Ali',  'grade': 'A'},
    {'id': 2, 'name': 'Sara', 'grade': 'B'},
]
next_id = 3
VALID_GRADES = {'A', 'B', 'C', 'D', 'F'}

# --- HEALTH ---
@app.route('/api/health')
def health():
    return jsonify({'status': 'ok', 'message': 'Flask is running'}), 200

# --- READ ALL ---
@app.route('/api/students', methods=['GET'])
def get_students():
    return jsonify(students), 200

# --- READ ONE ---
@app.route('/api/students/<int:sid>', methods=['GET'])
def get_student(sid):
    s = next((s for s in students if s['id'] == sid), None)
    return (jsonify(s), 200) if s else (jsonify({'error': 'Not found'}), 404)

# --- CREATE ---
@app.route('/api/students', methods=['POST'])
def add_student():
    global next_id
    data = request.get_json()
    if not data or 'name' not in data or 'grade' not in data:
        return jsonify({'error': 'name and grade required'}), 400
    if data['grade'] not in VALID_GRADES:
        return jsonify({'error': 'Invalid grade'}), 400
    new = {'id': next_id, 'name': data['name'], 'grade': data['grade']}
    students.append(new)
    next_id += 1
    return jsonify(new), 201

# --- UPDATE (full replace) ---
@app.route('/api/students/<int:sid>', methods=['PUT'])
def update_student(sid):
    s = next((s for s in students if s['id'] == sid), None)
    if not s:
        return jsonify({'error': 'Not found'}), 404
    data = request.get_json()
    s['name']  = data.get('name',  s['name'])
    s['grade'] = data.get('grade', s['grade'])
    return jsonify(s), 200

# --- PATCH (partial update) ---
@app.route('/api/students/<int:sid>', methods=['PATCH'])
def patch_student(sid):
    s = next((s for s in students if s['id'] == sid), None)
    if not s:
        return jsonify({'error': 'Not found'}), 404
    data = request.get_json() or {}
    if 'name'  in data: s['name']  = data['name']
    if 'grade' in data:
        if data['grade'] not in VALID_GRADES:
            return jsonify({'error': 'Invalid grade'}), 400
        s['grade'] = data['grade']
    return jsonify(s), 200

# --- DELETE ---
@app.route('/api/students/<int:sid>', methods=['DELETE'])
def delete_student(sid):
    global students
    orig = len(students)
    students = [s for s in students if s['id'] != sid]
    if len(students) == orig:
        return jsonify({'error': 'Not found'}), 404
    return jsonify({'message': 'Deleted'}), 200

# --- COUNT ---
@app.route('/api/students/count')
def count_students():
    return jsonify({'count': len(students)}), 200

# --- NAMES ONLY ---
@app.route('/api/students/names')
def get_names():
    return jsonify([s['name'] for s in students]), 200

# --- TOP (grade A only) ---
@app.route('/api/students/top')
def top_students():
    return jsonify([s for s in students if s['grade'] == 'A']), 200

# --- SEARCH by grade ---
@app.route('/api/students/search')
def search_students():
    grade = request.args.get('grade')
    if not grade:
        return jsonify({'error': 'grade param required'}), 400
    return jsonify([s for s in students if s['grade'] == grade]), 200

# --- STATS ---
@app.route('/api/students/stats')
def student_stats():
    counts = {}
    for s in students:
        counts[s['grade']] = counts.get(s['grade'], 0) + 1
    return jsonify({'total': len(students), 'by_grade': counts}), 200

if __name__ == '__main__':
    app.run(debug=True, port=5000)

    {'id': 2, 'name': 'Yahya', 'grade': 'A'},

    return jsonify(students)

def get_student(s_id):
    s = next((s for s in students if s['id'] == s_id), None)
    # Changed error code to 404 for 'Not Found'
    return (jsonify(s), 200) if s else (jsonify({'error': 'Not found'}), 404)

@app.route('/api/students', methods=['POST'])
def add_student():
    global next_id
    valid_grade = ['A','B','C','D','F']

    data = request.get_json()
    
    if not data or 'name' not in data or 'grade' not in data:
        return jsonify({'error': 'Data not complete'}), 400 # 400 is better for Bad Request

    if data['grade'].upper() not in valid_grade:
        return jsonify({'error' : 'Invalid grade'}) ,400
    new_student = {
        'id': next_id, 
        'name': data['name'], 
        'grade': data['grade']
    }
    students.append(new_student)
    next_id += 1

    return jsonify(new_student), 201 # 201 Created is standard for POST

@app.route('/api/students/count', methods = ['GET'])
def student_count():    
    return jsonify({"count" : len(students)}), 200

@app.route('/api/students/search' , methods=['GET'])

    grade_query = request.args.get('grade')
    if not grade_query:
        return jsonify({'error' : 'Provide a grade'}), 400
    filtered = [s for s in students if s.get('grade',' ').upper() == grade_query.upper()]
    return jsonify(filtered), 200
